# The Ubuntu 24.04 packages for making the Linux dists, read with apt-packages (in this directory).
# Used by the Dockerfile here, and by the GitHub runners (.github/workflows/dist.yml).
#
# [tools]: tools that run on the build machine. Always installed without an architecture suffix.
# [libraries]: development packages for the dist's architecture, as plain names. The Dockerfile
# installs them for the image's own architecture; the GitHub runners (and cross-compiling) append
# :amd64 or :arm64 themselves.
#
# SDL3, SDL3_image, SDL3_ttf and Python come from the env recipes (the [build] packages in
# od-fs/build/env/linux.txt), built into the env prefix at dist time; everything else comes from
# these packages. There are no development packages for the Python modules on the purge list
# (tk-dev, libgdbm-dev, libreadline-dev, libncurses-dev), so the env Python is built without them.
#
# One package per line; "#" starts a comment.

[tools]
autoconf
automake
build-essential
ca-certificates # downloads by the env system
cmake           # SDL3, SDL3_image, SDL3_ttf (env)
curl            # downloads by the env system

file
gettext         # msgfmt (configure)
git
libtool
patchelf        # rpaths in the dist
pkgconf
python3         # env system scripts, the dist check (3.12+)
tar
